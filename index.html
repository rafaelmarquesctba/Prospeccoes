const SHEET_ID = '1C9tdinvjiRGohExVBrt6QhwjhQgRpKV1WnxnS6bVeVs'; const SHEET_NAME = 'CONTROLE DE VISITAS'; function doGet(e) { const action = e.parameter.action; const callback = e.parameter.callback; let result; if (action === 'getAll') result = getAll(); else if (action === 'search') result = search(e.parameter.q); else result = { error: 'Ação inválida' }; return respond(result, callback); } function doPost(e) { const data = JSON.parse(e.postData.contents); let result; if (data.action === 'update') result = updateRow(data); else if (data.action === 'insert') result = insertRow(data); else if (data.action === 'createEvent') result = createEvent(data); else result = { error: 'Ação inválida' }; return respond(result, null); } function respond(data, callback) { const json = JSON.stringify(data); if (callback) { return ContentService .createTextOutput(callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function getSheet() { return SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); } function rowToObj(row, index) { return { rowIndex: index, infoVia: row[1] || '', primeiroContato: row[2] ? Utilities.formatDate(new Date(row[2]), 'America/Sao_Paulo', 'dd/MM/yyyy') : '', cliente: row[3] || '', ultimoContato: row[4] ? Utilities.formatDate(new Date(row[4]), 'America/Sao_Paulo', 'dd/MM/yyyy') : '', endereco: row[5] || '', cidade: row[6] || '', telefone: row[7] || '', contato: row[8] || '', funcao: row[9] || '', email: row[10] || '', status: row[11] || '', observacoes: row[12] || '', }; } function getAll() { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const result = []; for (let i = 1; i < data.length; i++) { if (data[i][3]) result.push(rowToObj(data[i], i + 1)); } return result; } function search(q) { if (!q) return getAll(); const lower = q.toLowerCase(); return getAll().filter(r => r.cliente.toLowerCase().includes(lower) || r.contato.toLowerCase().includes(lower) || r.cidade.toLowerCase().includes(lower) || r.telefone.toLowerCase().includes(lower) ); } function parseDate(str) { if (!str) return ''; const parts = str.split('/'); if (parts.length === 3) return new Date(parts[2], parts[1] - 1, parts[0]); return new Date(str); } function updateRow(data) { const sheet = getSheet(); const row = data.rowIndex; if (data.ultimoContato) sheet.getRange(row, 5).setValue(parseDate(data.ultimoContato)); if (data.endereco) sheet.getRange(row, 6).setValue(data.endereco); if (data.cidade) sheet.getRange(row, 7).setValue(data.cidade); if (data.telefone) sheet.getRange(row, 8).setValue(data.telefone); if (data.contato) sheet.getRange(row, 9).setValue(data.contato); if (data.funcao) sheet.getRange(row, 10).setValue(data.funcao); if (data.email) sheet.getRange(row, 11).setValue(data.email); if (data.status) sheet.getRange(row, 12).setValue(data.status); if (data.observacoes !== undefined) sheet.getRange(row, 13).setValue(data.observacoes); return { success: true }; } function insertRow(data) { const sheet = getSheet(); const lastRow = sheet.getLastRow() + 1; sheet.getRange(lastRow, 1).setValue(1); sheet.getRange(lastRow, 2).setValue(data.infoVia); sheet.getRange(lastRow, 3).setValue(new Date()); sheet.getRange(lastRow, 4).setValue(data.cliente); sheet.getRange(lastRow, 5).setValue(parseDate(data.ultimoContato)); sheet.getRange(lastRow, 6).setValue(data.endereco); sheet.getRange(lastRow, 7).setValue(data.cidade); sheet.getRange(lastRow, 8).setValue(data.telefone); sheet.getRange(lastRow, 9).setValue(data.contato); sheet.getRange(lastRow, 10).setValue(data.funcao); sheet.getRange(lastRow, 11).setValue(data.email); sheet.getRange(lastRow, 12).setValue(data.status); sheet.getRange(lastRow, 13).setValue(data.observacoes); return { success: true, rowIndex: lastRow }; } function createEvent(data) { const calendar = CalendarApp.getCalendarById(CALENDAR_ID) || CalendarApp.getDefaultCalendar(); const parts = data.date.split('-'); const timeParts = data.time.split(':'); const start = new Date(parts[0], parts[1]-1, parts[2], timeParts[0], timeParts[1], 0); const end = new Date(start.getTime() + data.duration * 60000); const options = { description: data.description || '', location: data.location || '', }; const event = calendar.createEvent(data.title, start, end, options); // Lembrete 1 dia antes (1440 minutos) event.addPopupReminder(1440); // Lembrete 1 hora antes também event.addPopupReminder(60); return { success: true, eventId: event.getId() }; }
